import yaml
import json
import os
from datetime import datetime
from apscheduler.schedulers.background import BackgroundScheduler

# 岗位数据存储文件
JOB_DATA_FILE = "job_record.json"

# 初始化存储文件
def init_job_file():
    if not os.path.exists(JOB_DATA_FILE):
        init_data = {
            "all_jobs": [],
            "apply_records": []
        }
        with open(JOB_DATA_FILE, "w", encoding="utf-8") as f:
            json.dump(init_data, f, ensure_ascii=False, indent=2)

# 读取配置
def load_config():
    with open("config.yaml", "r", encoding="utf-8") as f:
        return yaml.safe_load(f)

# 读取岗位数据库
def load_job_data():
    with open(JOB_DATA_FILE, "r", encoding="utf-8") as f:
        return json.load(f)

# 写入岗位数据库
def save_job_data(data):
    with open(JOB_DATA_FILE, "w", encoding="utf-8") as f:
        json.dump(data, ensure_ascii=False, indent=2)

# 合并岗位，自动去重
def merge_new_jobs():
    data = load_job_data()
    old_ids = [item["job_id"] for item in data["all_jobs"]]
    fresh_jobs = []
    for job in fetch_new_jobs():
        if job["job_id"] not in old_ids:
            data["all_jobs"].append(job)
            fresh_jobs.append(job)
    save_job_data(data)
    return fresh_jobs

# 计算截止剩余天数
def get_deadline_days(deadline_str):
    today = datetime.now()
    deadline = datetime.strptime(deadline_str, "%Y-%m-%d")
    diff = (deadline - today).days
    return diff

# 展示全部岗位清单
def show_all_jobs():
    data = load_job_data()
    job_list = data["all_jobs"]
    if not job_list:
        print("暂无岗位信息，等待下一轮自动更新")
        return
    print("\n===== 全部秋招岗位清单（每日自动更新）=====")
    for job in job_list:
        days = get_deadline_days(job["deadline_date"])
        warn = "【紧急！即将截止】" if days <=7 else ""
        print(f"\n岗位ID：{job['job_id']} {warn}")
        print(f"岗位：{job['title']} | 公司：{job['company']} | 薪资：{job['salary']}")
        print(f"发布时间：{job['publishing_date']} ｜网申截止：{job['deadline_date']} 剩余{days}天")
        print(f"投递网址：{job['apply_url']}")
        print(f"当前投递进度：{job['status']}")

# 修改岗位投递进度状态
def change_apply_status(job_id, new_status):
    data = load_job_data()
    for job in data["all_jobs"]:
        if job["job_id"] == job_id:
            job["status"] = new_status
            log_info = {
                "job_id": job_id,
                "operate_time": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                "new_status": new_status
            }
            data["apply_records"].append(log_info)
            save_job_data(data)
            print(f"✅ 岗位{job_id}状态更新成功，当前进度：{new_status}")
            return
    print(f"❌ 未找到ID为{job_id}的岗位")

# 问答匹配回复
def qa_reply(question, cfg):
    for keyword, answer in cfg["qa_rule"].items():
        if keyword in question:
            return answer
    return "暂未收录该问题，支持咨询秋招、档案、贵阳人才补贴、社保公积金相关内容"

# 每日定时更新岗位任务
def daily_update_job_task():
    print(f"\n[{datetime.now().strftime('%Y-%m-%d %H:%M')}] 执行每日岗位自动更新")
    fresh = merge_new_jobs()
    if fresh:
        print(f"今日抓取到{len(fresh)}条全新匹配秋招岗位！")
        for j in fresh:
            d = get_deadline_days(j["deadline_date"])
            tip = "⚠️截止临近" if d <=7 else ""
            print(f"- {j['title']} | 截止{j['deadline_date']} 剩余{d}天 {tip}")
    else:
        print("今日没有新增岗位")

# 程序启动入口
if __name__ == "__main__":
    init_job_file()
    config = load_config()
    print("秋招小助手已启动 ✅")
    print("可用指令：")
    print("1、立即更新岗位：手动刷新岗位库")
    print("2、查看全部岗位：展示岗位链接、截止时间、投递进度")
    print("3、修改状态：修改状态 岗位ID 状态名，例：修改状态 job001 已投递")
    print("4、问答：直接输入求职、档案、补贴问题")
    print("5、exit：关闭助手")

    # 定时任务：每天凌晨2点自动更新岗位
    scheduler = BackgroundScheduler()
    scheduler.add_job(daily_update_job_task, "cron", hour=2, minute=0)
    scheduler.start()

    while True:
        user_input = input("\n请输入指令：")
        if user_input.strip() == "exit":
            scheduler.shutdown()
            print("秋招助手停止运行，岗位数据已保存")
            break
        elif user_input == "立即更新岗位":
            daily_update_job_task()
        elif user_input == "查看全部岗位":
            show_all_jobs()
        elif user_input.startswith("修改状态"):
            parts = user_input.split()
            if len(parts)>=3:
                jid = parts[2]
                stat = parts[3]
                change_apply_status(jid, stat)
            else:
                print("格式错误！正确示例：修改状态 job001 已投递")
        else:
            res = qa_reply(user_input, config)
            print("助手回复：", res)
